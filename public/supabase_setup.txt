-- Enable UUID extension
create extension if not exists "uuid-ossp";

-- Table: VYAHE_ridercustomer_users
create table public.VYAHE_ridercustomer_users (
  id uuid references auth.users not null primary key,
  name text not null,
  email text not null,
  phone text,
  role text check (role in ('Customer', 'Rider', 'Admin')),
  is_online boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Enable RLS
alter table public.VYAHE_ridercustomer_users enable row level security;

-- Policies for VYAHE_ridercustomer_users
create policy "Public profiles are viewable by everyone"
  on public.VYAHE_ridercustomer_users for select
  using ( true );

create policy "Users can insert their own profile"
  on public.VYAHE_ridercustomer_users for insert
  with check ( auth.uid() = id );

create policy "Users can update own profile"
  on public.VYAHE_ridercustomer_users for update
  using ( auth.uid() = id );

-- Table: VYAHE_orders
create table public.VYAHE_orders (
  id uuid default uuid_generate_v4() primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  customer_id uuid references public.VYAHE_ridercustomer_users(id) not null,
  rider_id uuid references public.VYAHE_ridercustomer_users(id),
  pickup_address text not null,
  delivery_address text not null,
  pickup_lat double precision,
  pickup_lng double precision,
  delivery_lat double precision,
  delivery_lng double precision,
  items jsonb not null, -- Array of {name, qty, price}
  total_amount numeric not null,
  delivery_fee numeric not null,
  payment_method text check (payment_method in ('Cash', 'GCash')),
  payment_status text default 'pending',
  status text default 'pending_approval', -- pending_approval, approved, assigned, picked_up, delivered, completed, cancelled
  assigned_at timestamp with time zone,
  delivered_at timestamp with time zone
);

-- Enable RLS
alter table public.VYAHE_orders enable row level security;

-- Policies for VYAHE_orders
create policy "Customers can view their own orders"
  on public.VYAHE_orders for select
  using ( auth.uid() = customer_id );

create policy "Riders can view approved and assigned orders"
  on public.VYAHE_orders for select
  using ( 
    (status = 'approved') or 
    (rider_id = auth.uid()) 
  );

create policy "Admins can view all orders"
  on public.VYAHE_orders for select
  using ( 
    exists (
      select 1 from public.VYAHE_ridercustomer_users 
      where id = auth.uid() and role = 'Admin'
    )
  );

create policy "Customers can insert orders"
  on public.VYAHE_orders for insert
  with check ( auth.uid() = customer_id );

create policy "Riders can update orders they claimed"
  on public.VYAHE_orders for update
  using ( rider_id = auth.uid() or rider_id is null ); -- Simplified for claiming

create policy "Admins can update orders"
  on public.VYAHE_orders for update
  using ( 
    exists (
      select 1 from public.VYAHE_ridercustomer_users 
      where id = auth.uid() and role = 'Admin'
    )
  );

-- Table: VYAHE_messages
create table public.VYAHE_messages (
  id uuid default uuid_generate_v4() primary key,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  order_id uuid references public.VYAHE_orders(id) not null,
  sender_id uuid references public.VYAHE_ridercustomer_users(id) not null,
  receiver_id uuid references public.VYAHE_ridercustomer_users(id) not null,
  content text not null
);

-- Enable RLS
alter table public.VYAHE_messages enable row level security;

-- Policies for VYAHE_messages
create policy "Users can view messages they sent or received"
  on public.VYAHE_messages for select
  using ( auth.uid() = sender_id or auth.uid() = receiver_id );

create policy "Users can insert messages"
  on public.VYAHE_messages for insert
  with check ( auth.uid() = sender_id );

-- Table: VYAHE_settings
create table public.VYAHE_settings (
  id text primary key,
  price_per_km numeric default 20.0
